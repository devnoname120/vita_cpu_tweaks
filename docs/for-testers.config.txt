# vita_prefetch_enabler -- goes at ur0:data/prefetch_enabler.txt
#
# Step 1 (as shipped): probe_only=on -- read-only, makes NO secure writes.
# Step 2: probe_only=off, l2_prefetch=on
# Step 3: also l1_prefetch=on
# One reboot per step; read ur0:data/prefetch_enabler.log after each.
#
# LF, CRLF and bare CR line endings all work, as does a UTF-8 BOM, so it is
# safe to edit this on Windows, macOS or Linux. Values are "on" or anything
# else (= off). Lines starting with '#' are comments.

probe_only=on
l1_prefetch=off
l2_prefetch=off

# probe_only:   read-only recon -- dumps the secure page-table layout to the
#               log and makes no secure writes. Overrides the two below.
# l1_prefetch:  Cortex-A9 ACTLR L1 D-side prefetch + L2 prefetch hint
#               (bits 2 and 1)
# l2_prefetch:  ARM PL310 L2C-310 data prefetch (PREFETCH_CTRL bit 28)
